\section{CARACTERÍSTICAS (FEATURES)}
\label{sec:features}

\begin{itemize}[label=\small\color{IndustrialBlue}$\bullet$, leftmargin=1.2cm, itemsep=5pt]
    \item \textbf{Multifunción Digital:} Pantalla digital con alimentación por batería; permite mediciones de Inductancia, Capacitancia, Resistencia y prueba de Diodos.
    \item \textbf{Circuito LSI:} Incorpora circuitería de integración a gran escala (LSI) que garantiza alta confiabilidad y durabilidad operativa.
    \item \textbf{Visibilidad LCD:} Pantalla diseñada para lecturas claras incluso en condiciones de alta luminosidad ambiental.
    \item \textbf{Protección de Entrada:} Sistema integrado de protección contra sobrecarga en todos los rangos.
    \item \textbf{Selector Ergonómico:} Selector de funciones tipo rotativo para una operación simplificada y rápida.
    \item \textbf{Indicador de Estado:} Sistema de alerta de batería baja para prevenir errores de medición.
\end{itemize}

\section{PROCEDIMIENTO DE MEDICIÓN}
\label{sec:operacion}

\subsection{Medición de Inductancia (L)}

\begin{enumerate}[label=\protect\circled{\arabic*}, leftmargin=1.4cm, itemsep=5pt]
    \item \textbf{Encendido:} Deslice el interruptor (\textbf{3-2}) a la posición \textbf{``1''}.
    \item \textbf{Selección de Modo:} Deslice el selector \textbf{L/C, R} (\textbf{3-4}) a \textbf{``L/C''}.
    \item \textbf{Rango Inicial:} Gire el selector (\textbf{3-3}) al rango máximo esperado.
    \item \textbf{Conexión:} Conecte los clips (tipo cocodrilo) a la entrada (\textbf{3-5}) y sujete el inductor.
    \item \textbf{Lectura:} Observe la pantalla (\textbf{3-1}). Si aparece \textbf{``1''} (fuera de rango), seleccione el rango superior.
\end{enumerate}

\notastecnicas{CONSIDERACIONES TÉCNICAS DE MEDICIÓN}{%
    \begin{enumerate}[leftmargin=0.5cm, labelsep=0.2cm, itemsep=3pt]
        \item \textbf{Valores desconocidos:} Inicie en \textbf{2 mH} e incremente hasta lectura estable.
        \item \textbf{Límites de escala:} Rangos de 200 mH, 2 H y 20 H optimizados para valores $>$10\% del fondo de escala.
        \item \textbf{Compensación parásita (2 mH):} Cortocircuite los terminales, registre el residuo y réstelo de la lectura final.
        \item \textbf{Bajas inductancias:} Use cables de conexión extremadamente cortos para minimizar capacitancias parásitas.
        \item \textbf{Limitación:} El equipo mide inductancia pura, no el factor de calidad (\textbf{Q}).
    \end{enumerate}
}

\subsection{Medición de Capacitancia (C)}

\noindent
\begingroup
\color{red!80!black}
\hspace{0.5cm}\vrule width 1.5pt\hspace{10pt}%
\begin{minipage}{\dimexpr\textwidth-25pt-0.5cm\relax}
    \vspace{3pt}
    {\small\textbf{\faExclamationTriangle\ ADVERTENCIA DE SEGURIDAD}}\\
    \footnotesize\textbf{DESCARGA DE CAPACITORES:} Asegúrese de que todo capacitor esté \textbf{completamente descargado} antes de conectarlo. Un componente cargado puede dañar irreversiblemente los circuitos de protección.
    \vspace{3pt}
\end{minipage}
\endgroup

\vspace{6pt}

\begin{enumerate}[label=\small\color{IndustrialBlue}\bfseries\arabic*., leftmargin=1.4cm, itemsep=5pt]
    \item \textbf{Encendido:} Deslice el interruptor (\textbf{3-2}) a la posición \textbf{``1''}.
    \item \textbf{Selección de Modo:} Verifique que el selector (\textbf{3-4}) esté en \textbf{``L/C''}.
    \item \textbf{Rango Inicial:} Gire el selector (\textbf{3-3}) al rango máximo esperado.
    \item \textbf{Conexión y Polaridad:} Conecte los clips a la entrada (\textbf{3-5}). Respete la polaridad (+/--) en capacitores electrolíticos.
    \item \textbf{Lectura:} Si aparece \textbf{``1''} (fuera de rango), seleccione el rango superior.
\end{enumerate}

\notastecnicas{CONSIDERACIONES TÉCNICAS DE MEDICIÓN}{%
    \begin{enumerate}[leftmargin=0.5cm, labelsep=0.2cm, itemsep=3pt]
        \item \textbf{Valores desconocidos:} Inicie en \textbf{2 nF} e incremente hasta lectura válida.
        \item \textbf{Límites de escala:} El valor medido debe representar $>$10\% del fondo de escala para precisión nominal.
        \item \textbf{Compensación parásita (2 nF):} Clips abiertos, registre el residuo ($\approx$15 pF) y réstelo de la lectura.
        \item \textbf{Diagnóstico:} Fugas $\rightarrow$ valor elevado o sobre-rango. Abierto $\rightarrow$ cero o pocos pF.
        \item \textbf{Cables cortos:} Reduzca conexiones para minimizar capacitancias externas.
    \end{enumerate}
}

\subsection{Medición de Resistencia (R)}

\begin{enumerate}[label=\small\color{IndustrialBlue}\bfseries\arabic*., leftmargin=1.4cm, itemsep=5pt]
    \item \textbf{Encendido:} Deslice el interruptor (\textbf{3-2}) a la posición \textbf{``1''}.
    \item \textbf{Selección de Modo:} Deslice el selector (\textbf{3-4}) a la posición \textbf{``R''}.
    \item \textbf{Configuración de Rango:} Gire el selector (\textbf{3-3}) al rango máximo esperado.
    \item \textbf{Conexión:} Mediante clips en (\textbf{3-5}), o inserte directamente en (\textbf{3-6}).
    \item \textbf{Lectura:} Si aparece \textbf{``1''}, seleccione el rango superior. Use siempre el rango más bajo que permita lectura.
\end{enumerate}

\notastecnicas{CONSIDERACIONES DE PRECISIÓN}{%
    \begin{itemize}[leftmargin=0.4cm, labelsep=0.2cm, itemsep=3pt]
        \item \textbf{Compensación parásita:} En rangos bajos (20~$\Omega$ y 200~$\Omega$), cortocircuite los clips, registre el residuo y réstelo de la medición final.
    \end{itemize}
}

\begin{enumerate}[label=\small\color{IndustrialBlue}\bfseries\arabic*., leftmargin=1.4cm, itemsep=5pt]
    \item \textbf{Encendido:} Deslice el interruptor (\textbf{3-2}) a la posición \textbf{``1''}.
    \item \textbf{Selección de Modo:} Deslice el selector (\textbf{3-4}) a la posición \textbf{``R''}.
    \item \textbf{Rango de Diodo:} Gire el selector (\textbf{3-3}) a la posición \textbf{$\rightarrow|-$}.
    \item \textbf{Conexión:} Clip rojo al terminal positivo (+), clip negro al negativo (--).
    \item \textbf{Evaluación Completa:} Realice ambas pruebas para diagnóstico correcto.
\end{enumerate}

\notastecnicas{GUÍA DE DIAGNÓSTICO DE DIODOS}{%
    \begin{itemize}[leftmargin=0.4cm, labelsep=0.2cm, itemsep=4pt]

        \end{itemize}
        \item \textbf{B. Polarización Inversa} (Cátodo al rojo, Ánodo al negro):
        \begin{itemize}[label=$\circ$, nosep]
            \item \textbf{Bueno:} Pantalla muestra \textbf{``1''} (bloqueo de corriente).
            \item \textbf{Defectuoso:} Muestra ``.000'' u otros valores numéricos.
        \end{itemize}
    \end{itemize}

\section{MANTENIMIENTO}

\subsection{Consideraciones de Uso}
\begin{itemize}[label=\color{IndustrialBlue}$\bullet$, leftmargin=1.2cm, itemsep=5pt]
    \item \textbf{Limitaciones Técnicas:} Mide capacitancia e inductancia pura. No determina el factor de calidad (\textbf{Q}). Evite medir componentes reactivos en serie con resistencias.
    \item \textbf{Seguridad en Circuitos:} Asegúrese de que el circuito esté \textbf{completamente desenergizado} antes de conectar los clips.
    \item \textbf{Cuidado Ambiental:} Limpie periódicamente en entornos con polvo. Evite exposición prolongada al calor solar directo.
\end{itemize}

\subsection{Reemplazo de Batería}
\begin{enumerate}[label=\small\color{IndustrialBlue}\bfseries\arabic*., leftmargin=1.4cm, itemsep=5pt]
    \item \textbf{Indicador de Batería Baja:} Al aparecer \textbf{``LOBAT''} en la esquina superior izquierda (\textbf{3-1}), reemplace la batería. El equipo mantendrá precisión algunas horas más antes de dar lecturas erráticas.
    \item \textbf{Procedimiento:}
    \begin{itemize}[label=$\circ$, nosep]
        \item Apague el equipo y desconecte los componentes de los terminales.
        \item Retire el tornillo de la tapa de batería (\textbf{3-7}) con destornillador adecuado.
        \item Reemplace con batería \textbf{9V} (NEDA 1604 o equivalente) y recoloque la tapa.
    \end{itemize}
\end{enumerate}
