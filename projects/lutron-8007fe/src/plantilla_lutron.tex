\section{CARACTERÍSTICAS (FEATURES)}
\label{sec:features}

\begin{itemize}[label=\small\color{IndustrialBlue}$\bullet$, leftmargin=1cm, itemsep=4pt]
    \item \textbf{Multifunción Digital:} Pantalla digital con alimentación por batería; permite mediciones de Inductancia, Capacitancia, Resistencia y prueba de Diodos.

    \item \textbf{Protección de Entrada:} Sistema integrado de protección contra sobrecarga en todos los rangos.
    \item \textbf{Selector Ergonómico:} Selector de funciones tipo rotativo para una operación simplificada y rápida.
    \item \textbf{Indicador de Estado:} Sistema de alerta de batería baja para prevenir errores de medición.
\end{itemize}

\section{PROCEDIMIENTO DE MEDICIÓN}
\label{sec:operacion}

% --- 4.1 INDUCTANCIA ---
\subsection{Medición de Inductancia (L)}

\begin{enumerate}[label=\protect\circled{\arabic*}, leftmargin=1.2cm, itemsep=0.3cm]
    \item \textbf{Encendido:} Deslice el interruptor de encendido (\textbf{3-2}) a la posición \textbf{"1"}.
    \item \textbf{Selección de Modo:} Deslice el selector \textbf{L/C, R} (\textbf{3-4}) a la posición \textbf{"L/C"}.
    \item \textbf{Rango Inicial:} Gire el selector de función (\textbf{3-3}) hasta el rango máximo esperado de inductancia. 
    \item \textbf{Conexión:} Conecte los clips de prueba (tipo cocodrilo) a la entrada de medición (\textbf{3-5}) y luego sujete el inductor a medir.
    \item \textbf{Lectura:} Observe el valor en la pantalla (\textbf{3-1}). 
    \begin{itemize}[label=$\rightarrow$, nosep]
        \item Si aparece un \textbf{"1"} en el display, indica que la medición está fuera de rango (\textit{Out-of-Range}). Seleccione el siguiente rango superior para obtener una lectura.
    \end{itemize}
\end{enumerate}

\vspace{0.4cm}

\begin{minipage}{\dimexpr\textwidth-25pt-0.5cm\relax}
    \vspace{5pt}
    {\footnotesize \textbf{CONSIDERACIONES TÉCNICAS DE MEDICIÓN}} \\[0.2cm]
    \scriptsize
    \begin{enumerate}[leftmargin=0.4cm, labelsep=0.2cm, itemsep=0.1cm]
        \item \textbf{Valores desconocidos:} Si el inductor no posee marcado, inicie en el rango de \textbf{2 mH} e incremente la escala hasta obtener una lectura estable.
        
        \item \textbf{Límites de escala:} Los rangos de \textbf{200 mH}, \textbf{2 H} y \textbf{20 H} están optimizados para mediciones superiores al 10\% de su fondo de escala. Es normal obtener lecturas erráticas si se miden valores inferiores a dicho umbral (ej. menos de 20 mH en el rango de 200 mH).

        \item \textbf{Compensación de Inductancia Parásita (Rango 2 mH):} 
        Para mediciones de alta precisión, considere la inductancia residual del circuito interno:
        \begin{itemize}[label=$\circ$, nosep]
            \item Cortocircuite los terminales de entrada y registre el valor residual (ej. $-15\,\mu$H).
            \item Reste este valor de la lectura final. (\textit{Ejemplo: Si el residuo es $-15\,\mu$H, sume $15\,\mu$H al valor medido}).
        \end{itemize}

        \item \textbf{Bajas Inductancias:} Para evitar capacitancias parásitas, utilice cables de conexión extremadamente cortos.

        \item \textbf{Limitaciones:} Este equipo mide inductancia pura, no el factor de calidad (\textbf{Q}). Evite medir inductancia en resistencias, ya que obtendrá lecturas erróneas.
    \end{enumerate}
    \vspace{5pt}

% Bloque de Alerta de Seguridad (Prioridad Alta)

\begingroup

    {\footnotesize \textbf{\faExclamationTriangle\ ADVERTENCIA DE SEGURIDAD}} \\
    \scriptsize \textbf{DESCARGA DE CAPACITORES:} Asegúrese de que todo capacitor esté \textbf{completamente descargado} antes de conectarlo. Un componente cargado puede dañar de forma irreversible los circuitos de protección del equipo.
    \vspace{2pt}
\end{minipage}
\endgroup

\vspace{0.4cm}

\begin{enumerate}[label=\protect\small\color{IndustrialBlue}\bfseries\arabic*., leftmargin=1.2cm, itemsep=0.3cm]
    \item \textbf{Encendido:} Deslice el interruptor de encendido (\textbf{3-2}) a la posición \textbf{"1"}.
    \item \textbf{Selección de Modo:} Verifique que el selector \textbf{L/C, R} (\textbf{3-4}) esté en la posición \textbf{"L/C"}.
    \item \textbf{Rango Inicial:} Gire el selector de función (\textbf{3-3}) hacia el rango máximo esperado de capacitancia.
    \item \textbf{Conexión y Polaridad:} 
        \begin{itemize}[label=$\rightarrow$, nosep]
            \item Conecte los clips de prueba a la entrada de medición (\textbf{3-5}).
            \item \textbf{Nota crítica:} Observe la polaridad (+/-) al conectar capacitores electrolíticos o polarizados.
        \end{itemize}
    \item \textbf{Lectura:} Observe el valor en la pantalla (\textbf{3-1}). 
    \begin{itemize}[label=\color{IndustrialBlue}$\rightarrow$, nosep]
        \item Si aparece un \textbf{"1"} (fuera de rango), seleccione el rango inmediatamente superior para obtener la lectura.
    \end{itemize}
\end{enumerate}

% --- NOTAS TÉCNICAS DE MEDICIÓN (C) ---
\vspace{0.4cm}
\noindent
\begingroup
\color{DarkGray}
\hspace{0.5cm} \vrule width 1.5pt \hspace{10pt}
\begin{minipage}{\dimexpr\textwidth-25pt-0.5cm\relax}
    \vspace{5pt}
    {\footnotesize \textbf{CONSIDERACIONES TÉCNICAS DE MEDICIÓN}} \\[0.2cm]
    \scriptsize
    \begin{enumerate}[leftmargin=0.4cm, labelsep=0.2cm, itemsep=0.1cm]
        \item \textbf{Valores desconocidos:} Si el componente no está marcado, inicie en el rango de \textbf{2 nF} e incremente hasta obtener una lectura válida.
        
        \item \textbf{Límites de escala:} Para mantener la precisión nominal, elija el rango donde el valor medido represente más del 10\% del fondo de escala. Es normal obtener lecturas inconsistentes si mide valores por debajo de:
        \begin{itemize}[label=$\circ$, nosep]
            \item 2 nF en el rango de 200 nF | 0.02 $\mu$F en el rango de 2 $\mu$F.
            \item 0.2 $\mu$F en el rango de 20 $\mu$F | 2 $\mu$F en el rango de 200 $\mu$F.
        \end{itemize}

        \item \textbf{Compensación de Capacitancia Parásita (2 nF):} 
        Para alta precisión en valores bajos:
        \begin{itemize}[label=$\circ$, nosep]
            \item Mantenga los clips de prueba abiertos (sin conectar el componente) y registre el valor residual (ej. $15\,$pF).
            \item Reste este valor de la lectura final obtenida.
        \end{itemize}

        \item \textbf{Diagnóstico de Fallas:} 
        \begin{itemize}[label=$\circ$, nosep]
            \item \textbf{Fugas:} Un capacitor con fugas de voltaje mostrará un valor mucho más alto de lo normal o sobre-rango ("1").
            \item \textbf{Abierto:} Un componente en circuito abierto marcará cero en todos los rangos (o unos pocos pF en el rango de 2nF).
        \end{itemize}

        \item \textbf{Tolerancia y Cables:} Utilice conexiones cortas para evitar capacitancias parásitas externas. Recuerde que los capacitores electrolíticos suelen tener tolerancias muy amplias por diseño.
    \end{enumerate}
    \vspace{5pt}
\end{minipage}
\endgroup

\subsection{Medición de Resistencia (R)}

\begin{enumerate}[label=\protect\small\color{IndustrialBlue}\bfseries\arabic*., leftmargin=1.2cm, itemsep=0.3cm]
    \item \textbf{Encendido:} Deslice el interruptor de encendido (\textbf{3-2}) a la posición \textbf{"1"}.
    \item \textbf{Selección de Modo:} Deslice el selector \textbf{L/C, R} (\textbf{3-4}) a la posición \textbf{"R"}.
    \item \textbf{Configuración de Rango:} Gire el selector de función (\textbf{3-3}) hacia el rango máximo esperado.
    \item \textbf{Conexión del Componente:} 
        \begin{itemize}[label=$\rightarrow$, nosep]
            \item Conecte los clips de prueba a la entrada de medición (\textbf{3-5}) y sujete la resistencia.
            \item Alternativamente, inserte los terminales de la resistencia directamente en la \textbf{Entrada Directa (3-6)}.
        \end{itemize}
    \item \textbf{Lectura y Ajuste:} Observe el valor en la pantalla (\textbf{3-1}). 
    \begin{itemize}[label=\color{IndustrialBlue}$\rightarrow$, nosep]
        \item Si el display muestra \textbf{"1"}, seleccione el rango superior. Para optimizar la precisión, busque siempre el rango más bajo que permita una lectura.
    \end{itemize}
\end{enumerate}

% --- 8. MANTENIMIENTO ---
\section{MANTENIMIENTO}

\subsection{Consideraciones de Uso}
\begin{itemize}[label=\color{IndustrialBlue}$\bullet$, leftmargin=1cm]
    \item \textbf{Limitaciones Técnicas:} Este instrumento está diseñado para medir capacitancia e inductancia pura. No determina el factor de calidad (\textbf{Q}). Evite medir componentes reactivos en serie con resistencias, ya que las lecturas serán erróneas.
    \item \textbf{Seguridad en Circuitos:} Al medir componentes integrados en placas, asegúrese de que el circuito esté \textbf{completamente desenergizado} antes de conectar los clips de prueba.
    \item \textbf{Cuidado Ambiental:} En entornos con polvo, el equipo debe ser limpiado periódicamente. Evite la exposición prolongada al calor directo del sol para proteger la pantalla LCD y la carcasa.
\end{itemize}

\subsection{Reemplazo de Batería}
\begin{enumerate}[label=\protect\small\color{IndustrialBlue}\bfseries\arabic*., leftmargin=1.2cm]
    \item \textbf{Indicador de Batería Baja:} Cuando aparezca el aviso \textbf{"LOBAT"} en la esquina superior izquierda de la pantalla (\textbf{3-1}), es necesario reemplazar la batería. 
    \begin{itemize}[label=$\rightarrow$, nosep]
        \item \textit{Nota: El equipo mantendrá su precisión durante algunas horas después de la aparición del aviso antes de que las lecturas se vuelvan erráticas.}
    \end{itemize}
    \item \textbf{Procedimiento:} 
    \begin{itemize}[label=$\circ$, nosep]
        \item Apague el equipo y desconecte cualquier componente de los terminales de entrada.
        \item Retire el tornillo de la tapa de la batería (\textbf{3-7}) con un destornillador adecuado.
        \item Reemplace por una batería de \textbf{9V} (NEDA 1604 o equivalente) y vuelva a colocar la tapa.
    \end{itemize}
\end{enumerate}
